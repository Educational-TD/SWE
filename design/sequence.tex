	\noindent\textbf{Upgrade a Tower:}
	\begin{center}
		\begin{tikzpicture}
			\begin{umlseqdiag}
				\umlactor[class=Player]{player}
				\umlobject[class=Tower]{tower}
				
				\begin{umlcall}[op=upgradeTower(attribute)]{player}{tower}
					\umlcreatecall[class=Question]{tower}{question}
					
					\begin{umlcall}[op=Submit Answer, dt=10]{player}{tower}
					\end{umlcall}
					\begin{umlcall}[op=answer(response), dt=0]{tower}{question}
						\begin{umlfragment}[type=alt, label=Correct Answer, inner xsep=10, inner ysep=2]
							\begin{umlcall}[type=return, op=true]{question}{tower}
							\end{umlcall}
							\begin{umlcallself}[op=self.attribute++, dt=0]{tower}{tower}
							\end{umlcallself}
							\begin{umlcall}[type=return, op=true, dt=0, padding=0]{tower}{player}
							\end{umlcall}
							
							\umlfpart[Wrong Answer]
							\begin{umlcall}[type=return, op=false, dt=8]{question}{tower}
							\end{umlcall}
							\begin{umlcall}[type=return, op=false, dt=-5, padding=-8]{tower}{player}
							\end{umlcall}
						\end{umlfragment}
					\end{umlcall}
				\end{umlcall}
				
				\umlsdnode[dt=2]{player}
				\umlsdnode[dt=2]{tower}
			\end{umlseqdiag}
			
			\draw[draw=white, fill=white] (-0.2,-0.5) rectangle ++(0.4,-8);
			\draw[dotted] (0,-0.5) -- (0,-8.5);
